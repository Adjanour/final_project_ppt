% Introduction
\begin{frame}{Introduction}
    \begin{itemize}
        \item LLMs hallucinate, producing factually incorrect outputs
        \vspace{0.3cm}
        \item KGQA requires answers that follow real graph structure
        \vspace{0.3cm}
        \item Constrained decoding (GCR, DoG) guarantees validity but not relevance
        \vspace{0.3cm}
        \item A 3-hop question on Freebase can produce $\sim$24,000 structurally valid paths, yet only one is correct
    \end{itemize}
\end{frame}

% Problem Statement
\begin{frame}{Statement of Problem}
    \begin{itemize}
        \item GCR's trie is fixed before decoding starts, so it cannot adapt to question intent or the evolving chain
        \vspace{0.3cm}
        \item The LLM must filter semantically irrelevant paths using internal knowledge, the same mechanism that causes hallucinations
    \end{itemize}
\end{frame}

% Objectives
\begin{frame}{Project Objectives (1/2)}
    \begin{itemize}
        \item Characterize the permissiveness problem by measuring SIR of GCR's static KG-Tries on WebQSP
        \vspace{0.3cm}
        \item Design a symbolic constraint oracle (TypeOracle) with range and type gates, avoiding embedding cost and threshold sensitivity
    \end{itemize}
\end{frame}

\begin{frame}{Project Objectives (2/2)}
    \begin{itemize}
        \item Implement DCA-Trie v1 (static filtering at construction time, FNR $<$ 5\%) and v2 (dynamic expansion per committed triplet)
        \vspace{0.3cm}
        \item Evaluate both against GCR baseline on WebQSP: Hits@1, F1, structural faithfulness, SIR, trie size
    \end{itemize}
\end{frame}

\begin{frame}{Upgraded Constraint}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
        \textbf{Baseline (GCR/DoG):}
        \[
          W_{\text{val}}^{\text{GCR}}(t) = f(G, E_q)
        \]
        \end{column}
        \begin{column}{0.48\textwidth}
        \textbf{Proposed (DCA-Trie):}
        \[
          W_{\text{val}}^{\text{DCA}}(t) = f(G, E_q, q, y_{<t})
        \]
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Design Conditions}
    \begin{itemize}
      \item \textbf{F (Faithfulness):} 100\% structural grounding in KG triplets
      \item \textbf{R (Relevance):} Lower SIR than baseline
      \item \textbf{P (Recall):} FNR on gold paths $<$ 5\%
    \end{itemize}
\end{frame}

% Constrained Decoding Math
\begin{frame}{Constrained Decoding in LLMs}
  At each token step $t$, the LLM produces logits $\mathbf{h}_t \in \mathbb{R}^{|V|}$ over the vocabulary $V$.
  A constraint mask $\mathbf{m}_t \in \{0,1\}^{|V|}$ zeros out invalid tokens:

  \[
    \tilde{P}(x_t) = \text{softmax}\!\bigl(\mathbf{h}_t + (1 - \mathbf{m}_t) \cdot (-\infty)\bigr)
  \]

  \vspace{0.5em}
  \begin{itemize}
    \item In GCR/DoG, $\mathbf{m}_t$ comes from a static trie built before decoding: $\mathbf{m}_t = f(G, E_q)$
    \item DCA-Trie conditions the mask on the question and partial output: $\mathbf{m}_t = f(G, E_q, q, y_{<t})$
    \item Only tokens corresponding to valid KG triples receive non-zero probability
  \end{itemize}
\end{frame}

% TypeOracle Design
\begin{frame}{TypeOracle: Gate Design}
  Two symbolic gates over KG ontology metadata, applied per candidate triple $(e_t, r, e')$:

  \vspace{0.5em}
  \textbf{Gate 1: Property Range Gate (every hop)}
  \[
    \text{range\_gate}(r, e') = \mathbf{1}\!\bigl[\text{types}(e') \cap \text{range}(r) \neq \emptyset\bigr]
  \]
  Checks that the tail entity's types fall within the relation's expected range.

  \vspace{0.5em}
  \textbf{Gate 2: Answer Type Gate (terminal hop only)}
  \[
    \text{type\_gate}(e', q, h, L) = \mathbf{1}\!\bigl[\text{types}(e') \cap \mathcal{T}(q) \neq \emptyset\bigr]
  \]
  Checks that the terminal entity matches the inferred answer type $\mathcal{T}(q)$.

  \vspace{0.5em}
  \textbf{Combined:} $\text{admissible} = \text{range\_gate} \land \text{type\_gate}$, with conservative fallback (admit when schema info is missing).
\end{frame}
% SIR Definition
\begin{frame}{Diagnostic Metric: SIR}
  SIR measures the fraction of structurally valid paths that are semantically incompatible with the question.

  \[
    \text{SIR}(q, t) = \frac{\sum_{p \in \mathcal{P}(q, t)} \text{irrelevant}(p, q)}{|\mathcal{P}(q, t)|}
  \]

  \vspace{0.5em}
  \textbf{Corpus average:}
  \[
    \overline{\text{SIR}} = \frac{1}{|Q|} \sum_{q \in Q} \frac{1}{L_q} \sum_{t=1}^{L_q} \text{SIR}(q, t)
  \]
\end{frame}

% DCA-Trie v1
\begin{frame}{DCA-Trie v1: Static Filtering}
  \begin{itemize}
    \item Filters BFS-enumerated paths through TypeOracle before building the prefix tree
    \item Preserves $O(d)$ trie lookup while pruning irrelevant branches offline
  \end{itemize}
\end{frame}

% DCA-Trie v2
\begin{frame}{DCA-Trie v2: Dynamic Expansion}
  \begin{itemize}
    \item Expands the trie step-by-step as entity tokens $e_t$ are committed
    \item Applies range and type gate checks to $\mathcal{N}(e_t)$ during generation
  \end{itemize}
\end{frame}

% Discussion
\begin{frame}{Discussion}
    \begin{itemize}
        \item \textbf{Non-monotone:} Tighter constraints do not guarantee better answers.
        \vspace{0.3cm}
        \item \textbf{Four degradation mechanisms:} gold-path exclusion, beam competition, regex type-inference noise, precision--recall trade-off.
        \vspace{0.3cm}
        \item \textbf{TypeOracle efficiency:} $O(1)$ per path, no encoder, no threshold, only +0.5\% latency.
        \vspace{0.3cm}
        \item \textbf{SIR fills a gap:} diagnoses constraint quality independently of answer accuracy.
    \end{itemize}
\end{frame}

%Conclusion
\begin{frame}{Conclusion}
    \begin{itemize}
        \item DCA-Trie v1 meets all three conditions: 100\% faithfulness, 14.5\% path reduction, FNR $<$ 5\%.
        \vspace{0.3cm}
        \item TypeOracle achieves meaningful tightening with $O(1)$ symbolic gates.
        \vspace{0.3cm}
        \item Central finding: constraint tightness and accuracy are \textbf{non-monotone}, with Hits@1 dropping 5.2pp.
        \vspace{0.3cm}
        \item SIR is introduced as the first diagnostic for constraint quality, decoupled from accuracy.
    \end{itemize}
\end{frame}

%Recommendation
\begin{frame}{Recommendations}
    \begin{itemize}
        \item Replace regex type classifier with LLM-based intent parser (85\% $\to$ higher recall).
        \vspace{0.3cm}
        \item Extend TypeOracle schema beyond 40 Freebase relations.
        \vspace{0.3cm}
        \item Explore hybrid oracles (symbolic + embedding) to recover beam diversity.
        \vspace{0.3cm}
        \item Evaluate on CWQ with $N \geq 500$ to reduce variance.
    \end{itemize}
\end{frame}
