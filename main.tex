% Introduction
\begin{frame}{Introduction}
    \begin{itemize}
        \item LLMs hallucinate, producing plausible but factually incorrect outputs (Brown et al., 2020; Ji et al., 2023).
        \vspace{0.3cm}
        \item KGQA requires answers grounded in real graph structure (He et al., 2021; Lan et al., 2023).
        \vspace{0.3cm}
        \item Constrained decoding methods (GCR, DoG) guarantee structural validity but not semantic relevance (Geng et al., 2023; Li et al., 2025; Luo et al., 2025).
        \vspace{0.3cm}
        \item A 3-hop question on Freebase can produce $\sim$24,000 structurally valid paths, yet only one leads to the correct answer.
    \end{itemize}
\end{frame}

% Problem Statement
\begin{frame}{Statement of Problem}
    \begin{itemize}
        \item GCR's trie is fixed before decoding starts, so it cannot adapt to question intent or the evolving reasoning chain (Luo et al., 2025).
        \vspace{0.3cm}
        \item This forces the LLM to filter semantically irrelevant paths using internal knowledge, the same mechanism that causes hallucinations (Li et al., 2025).
        \vspace{0.3cm}
        \item Existing work assumes tighter constraints improve accuracy, but no metric exists to measure constraint quality independently of answer quality.
    \end{itemize}
\end{frame}

% Objectives
\begin{frame}{Project Objectives (1/2)}
    \begin{itemize}
        \item Characterize the permissiveness problem by measuring SIR of GCR's static KG-Tries on WebQSP
        \vspace{0.3cm}
        \item Design a symbolic constraint oracle (TypeOracle) with range and type gates, avoiding embedding cost and threshold sensitivity
    \end{itemize}
\end{frame}

\begin{frame}{Project Objectives (2/2)}
    \begin{itemize}
        \item Implement DCA-Trie v1 (static filtering at construction time, FNR $<$ 5\%) and v2 (dynamic expansion per committed triplet)
        \vspace{0.3cm}
        \item Evaluate both against GCR baseline on WebQSP: Hits@1, F1, structural faithfulness, SIR, trie size
    \end{itemize}
\end{frame}

% Constraint Specification
\begin{frame}{Upgraded Constraint}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Baseline (GCR/DoG):}
      \[
        W_{\text{val}}^{\text{GCR}}(t) = f(G, E_q)
      \]
      \vspace{0.3em}
      \footnotesize The valid token set depends only on the knowledge graph $G$ and entity mentions $E_q$. The same set of paths is available at every decoding step.
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Proposed (DCA-Trie):}
      \[
        W_{\text{val}}^{\text{DCA}}(t) = f(G, E_q, q, y_{<t})
      \]
      \vspace{0.3em}
      \footnotesize The valid token set also depends on the question $q$ and partial output $y_{<t}$. The constraint adapts as the model generates.
    \end{column}
  \end{columns}

  \vspace{0.5em}
  \centering
  \small\textit{Key difference: DCA-Trie conditions the mask on the question and partial output, enabling context-aware pruning.}
\end{frame}

\begin{frame}{Design Conditions}
    \begin{itemize}
      \item \textbf{F (Faithfulness):} 100\% structural grounding in KG triplets
      \item \textbf{R (Relevance):} Lower SIR than baseline
      \item \textbf{P (Recall):} FNR on gold paths $<$ 5\%
    \end{itemize}
\end{frame}

% Constrained Decoding Math
\begin{frame}{Constrained Decoding in LLMs}
  At each step $t$, the LLM produces logits $\mathbf{h}_t$ over vocabulary $V$. A mask $\mathbf{m}_t$ zeros out invalid tokens:

  \[
    \tilde{P}(x_t) = \text{softmax}\!\bigl(\mathbf{h}_t + (1 - \mathbf{m}_t) \cdot (-\infty)\bigr)
  \]

  \vspace{0.5em}
  \textbf{GCR/DoG (static):} $\mathbf{m}_t = f(G, E_q)$, trie built before decoding (Luo et al., 2025; Li et al., 2025).

  \vspace{0.3em}
  \textbf{DCA-Trie (dynamic):} $\mathbf{m}_t = f(G, E_q, q, y_{<t})$, adapts to question and partial output.
\end{frame}

% TypeOracle Design
\begin{frame}{TypeOracle: Gate Design}
  Two symbolic gates per candidate triple $(e_t, r, e')$, no encoder or threshold required:

  \vspace{0.5em}
  \textbf{Range Gate (every hop):} checks $e'$'s types against relation $r$'s range:
  \[
    \text{range\_gate}(r, e') = \mathbf{1}\!\bigl[\text{types}(e') \cap \text{range}(r) \neq \emptyset\bigr]
  \]

  \vspace{0.5em}
  \textbf{Type Gate (terminal hop):} checks terminal entity matches answer type $\mathcal{T}(q)$:
  \[
    \text{type\_gate}(e', q, h, L) = \mathbf{1}\!\bigl[\text{types}(e') \cap \mathcal{T}(q) \neq \emptyset\bigr]
  \]

  \vspace{0.5em}
  Admissible only if both pass. Unknown schema info conservatively admits the candidate.
\end{frame}

% SIR Definition
\begin{frame}{Diagnostic Metric: SIR}
  \textbf{SIR} measures the fraction of candidate paths that are \emph{semantically irrelevant} to the question, paths that exist in the knowledge graph but do not actually help answer the question.

  \[
    \text{SIR}(q, t) = \frac{\sum_{p \in \mathcal{P}(q, t)} \text{irrelevant}(p, q)}{|\mathcal{P}(q, t)|}
  \]

  \vspace{0.5em}
  \textbf{Corpus average:}
  \[
    \overline{\text{SIR}} = \frac{1}{|Q|} \sum_{q \in Q} \frac{1}{L_q} \sum_{t=1}^{L_q} \text{SIR}(q, t)
  \]

  \vspace{0.5em}
  A lower SIR means the oracle admits fewer irrelevant paths. GCR: $\overline{\text{SIR}} = 1.0$ (all paths admitted); DCA-Trie v1: $\overline{\text{SIR}} = 0.145$.
\end{frame}

% DCA-Trie v1
\begin{frame}{DCA-Trie v1: Static Filtering}
  \begin{itemize}
    \item Filters BFS-enumerated paths through TypeOracle before building the prefix tree
    \item Preserves $O(d)$ trie lookup while pruning irrelevant branches offline
  \end{itemize}
\end{frame}

% DCA-Trie v2
\begin{frame}{DCA-Trie v2: Dynamic Expansion}
  \begin{itemize}
    \item Expands the trie step-by-step as entity tokens $e_t$ are committed
    \item Applies range and type gate checks to $\mathcal{N}(e_t)$ during generation
  \end{itemize}
\end{frame}

\begin{frame}{Experimental Results: Runtime \& Memory}
  \begin{itemize}
    \item \textbf{Runtime:} DCA-Trie v1 adds $<$0.5\% latency over GCR (negligible overhead from $O(1)$ set intersection).
    \vspace{0.3cm}
    \item \textbf{Memory:} TypeOracle stores ontology metadata ($<$2 MB for Freebase); no additional GPU memory required.
    \vspace{0.3cm}
    \item \textbf{Scaling:} Gate cost is constant per candidate triple; total cost scales with BFS frontier size, not path length.
  \end{itemize}

  \vspace{0.5em}
  \textbf{Interpretation:} The symbolic oracle introduces no measurable computational bottleneck, making it practical for deployment.
\end{frame}

    \midrule
    \textbf{SIR Metric} & Accuracy only & Semantic Irrelevance Ratio & Diagnoses constraint quality independently \\
    \midrule
    \textbf{Non-Monotone} & ``Tighter $\rightarrow$ better'' & Tightness $\neq$ accuracy & First evidence constraint tightness \\
    \textbf{Finding} & (GCR, DoG assumption) & ($\Delta$Acc $=$ $-$5.2pp) & is not a proxy for quality \\
    \bottomrule
  \end{tabular}%
  }
\end{frame}

% Discussion
\begin{frame}{Discussion}
  \begin{itemize}
    \item TypeOracle reduced SIR by 85.5\% while maintaining structural faithfulness and FNR $<$ 5\%, yet Hits@1 dropped by 5.2pp; improvements in constraint quality did not translate into higher answer accuracy.
    \vspace{0.3cm}
    \item This challenges the implicit assumption in GCR and DoG that tightening the search space necessarily improves reasoning performance.
    \vspace{0.3cm}
    \item Structural constraint quality and reasoning accuracy are distinct properties that should be optimised independently.
    \vspace{0.3cm}
    \item Four degradation mechanisms identified: gold-path exclusion, beam competition, regex type-inference noise, and the precision--recall trade-off.
    \vspace{0.3cm}
    \item SIR provides the first diagnostic for measuring constraint quality decoupled from answer accuracy.
  \end{itemize}
\end{frame}

%Conclusion
\begin{frame}{Conclusion}
  \begin{itemize}
    \item DCA-Trie v1 achieves 100\% faithfulness, 14.5\% path reduction, and FNR $<$ 5\%, validating that symbolic gates meaningfully tighten the search space at negligible latency cost.
    \vspace{0.3cm}
    \item However, Hits@1 dropped 5.2pp, demonstrating that constraint tightness and answer accuracy are non-monotone.
    \vspace{0.3cm}
    \item Future work should pursue constraint quality and reasoning accuracy as separate optimisation targets, rather than assuming tighter constraints yield better answers.
    \vspace{0.3cm}
    \item SIR enables systematic comparison of oracles without conflating structural filtering with downstream accuracy.
  \end{itemize}
\end{frame}

%Recommendation
\begin{frame}{Recommendations}
    \begin{itemize}
        \item Replace regex type classifier with LLM-based intent parser (85\% $\to$ higher recall).
        \vspace{0.3cm}
        \item Extend TypeOracle schema beyond 40 Freebase relations.
        \vspace{0.3cm}
        \item Explore hybrid oracles (symbolic + embedding) to recover beam diversity.
        \vspace{0.3cm}
        \item Evaluate on CWQ with $N \geq 500$ to reduce variance.
    \end{itemize}
\end{frame}
